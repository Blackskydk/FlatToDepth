cmake_minimum_required(VERSION 3.24)
project(FlatToDepth LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "FlatToDepth requires Windows, MSVC and an x64 build.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OPENXR_ROOT "${CMAKE_SOURCE_DIR}/.deps/openxr" CACHE PATH "Extracted official OpenXR.Loader NuGet package")
find_path(OPENXR_INCLUDE openxr/openxr.h PATHS "${OPENXR_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
find_library(OPENXR_LIB openxr_loader PATHS "${OPENXR_ROOT}/native/x64/release/lib" REQUIRED NO_DEFAULT_PATH)
add_library(ftd_common INTERFACE)
target_include_directories(ftd_common INTERFACE src)
target_compile_definitions(ftd_common INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(ftd_common INTERFACE /W4 /permissive- /EHsc)
target_link_libraries(ftd_common INTERFACE d3d11 dxgi)
add_executable(FlatToDepth src/main.cpp)
target_include_directories(FlatToDepth PRIVATE "${OPENXR_INCLUDE}")
target_compile_definitions(FlatToDepth PRIVATE XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11)
target_link_libraries(FlatToDepth PRIVATE ftd_common "${OPENXR_LIB}")
add_custom_command(TARGET FlatToDepth POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OPENXR_ROOT}/native/x64/release/bin/openxr_loader.dll" "$<TARGET_FILE_DIR:FlatToDepth>")
enable_testing()
add_executable(FlatToDepthGpuSmoke tests/gpu_smoke.cpp)
target_link_libraries(FlatToDepthGpuSmoke PRIVATE ftd_common)
add_test(NAME gpu_share_split COMMAND FlatToDepthGpuSmoke)
add_executable(FlatToDepthCaptureProbe src/capture_probe.cpp)
target_link_libraries(FlatToDepthCaptureProbe PRIVATE ftd_common)
add_executable(FlatToDepthControllerTest tests/controller_test.cpp)
target_include_directories(FlatToDepthControllerTest PRIVATE "${OPENXR_INCLUDE}")
target_link_libraries(FlatToDepthControllerTest PRIVATE ftd_common)
add_test(NAME controller_transport_window COMMAND FlatToDepthControllerTest)
set_tests_properties(controller_transport_window PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}")
add_executable(FlatToDepthWindowTest tests/window_control_test.cpp)
target_include_directories(FlatToDepthWindowTest PRIVATE "${OPENXR_INCLUDE}")
target_link_libraries(FlatToDepthWindowTest PRIVATE ftd_common)
add_test(NAME window_pointer_control COMMAND FlatToDepthWindowTest)
add_executable(FlatToDepthGameMenuTest tests/game_menu_test.cpp)
target_include_directories(FlatToDepthGameMenuTest PRIVATE "${OPENXR_INCLUDE}")
target_link_libraries(FlatToDepthGameMenuTest PRIVATE ftd_common)
add_test(NAME game_menu COMMAND FlatToDepthGameMenuTest)
add_executable(FlatToDepthToolsTest tests/tools_test.cpp)
target_include_directories(FlatToDepthToolsTest PRIVATE "${OPENXR_INCLUDE}")
target_link_libraries(FlatToDepthToolsTest PRIVATE ftd_common)
add_test(NAME tools_panel_fx COMMAND FlatToDepthToolsTest "${CMAKE_SOURCE_DIR}")
add_executable(FlatToDepthCatalogTest tests/catalog_test.cpp)
target_include_directories(FlatToDepthCatalogTest PRIVATE "${OPENXR_INCLUDE}")
target_link_libraries(FlatToDepthCatalogTest PRIVATE ftd_common)
add_test(NAME games_catalog_scan COMMAND FlatToDepthCatalogTest "${CMAKE_SOURCE_DIR}")
